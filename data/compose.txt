================== DOCKER COMPOSE ==================

-- Subir / derrubar --
docker compose up                 # sobe os serviços
docker compose up -d              # em segundo plano
docker compose up -d --build      # rebuild antes de subir
docker compose down               # para e remove containers/redes
docker compose down -v            # também remove volumes

-- Controle --
docker compose start / stop / restart
docker compose ps                 # status dos serviços
docker compose logs -f SERVICO    # acompanha logs
docker compose exec SERVICO sh    # shell no serviço
docker compose run --rm SERVICO cmd   # comando avulso
docker compose build              # build das imagens
docker compose pull               # baixa imagens atualizadas
docker compose config             # valida e mostra o yaml final
docker compose -f outro.yml up    # usa outro arquivo

-- Exemplo de docker-compose.yml --
services:
  web:
    image: nginx:latest
    ports:
      - "8080:80"
    depends_on:
      - db
  db:
    image: postgres:16
    environment:
      POSTGRES_PASSWORD: senha
    volumes:
      - dados:/var/lib/postgresql/data
